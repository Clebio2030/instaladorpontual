INICIOCOMANDO
EXECUTE BLOCK
AS
BEGIN
  IF (NOT EXISTS(SELECT 1 FROM RDB$RELATIONS
                  WHERE RDB$RELATION_NAME = 'TELAS_USO_DIA'
                    AND COALESCE(RDB$SYSTEM_FLAG, 0) = 0)) THEN
    EXECUTE STATEMENT
      'CREATE TABLE TELAS_USO_DIA (' ||
      'DATA_USO DATE NOT NULL, ' ||
      'CLASSE_TELA VARCHAR(120) NOT NULL, ' ||
      'QTDE_ACESSOS INTEGER NOT NULL)';
END;
FIMCOMANDO
